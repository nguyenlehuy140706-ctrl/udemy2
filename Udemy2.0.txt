<script>
// Tìm kiếm khóa học
document.getElementById("search").addEventListener("input", function() {
    const keyword = this.value.toLowerCase();
    const courses = document.querySelectorAll(".course");

    courses.forEach(function(course) {
        const name = course.dataset.name.toLowerCase();
        course.style.display = name.includes(keyword) ? "" : "none";
    });
});

// Mô phỏng chức năng đăng ký
function registerCourse(courseName) {
    alert("Bạn đã chọn khóa học: " + courseName +
          "\nVui lòng đăng nhập để tiếp tục đăng ký.");
}
